cmake_minimum_required(VERSION 3.24)
project(CinderFoundry VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
option(FOUNDRY_BUILD_GUI "Build required C17 GTK desktop application" ON)
option(FOUNDRY_SANITIZE "Instrument Foundry with Address/Undefined sanitizers" OFF)
option(FOUNDRY_METAL "Enable llama.cpp Metal on macOS" OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/include/llama.h")
  message(FATAL_ERROR "Missing pinned backend: run git submodule update --init --recursive")
endif()
file(STRINGS docs/backend-revision.txt FOUNDRY_BACKEND_REVISION LIMIT_COUNT 1)
find_package(Git QUIET)
if(GIT_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp OUTPUT_VARIABLE actual_sha OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(actual_sha AND NOT actual_sha STREQUAL FOUNDRY_BACKEND_REVISION)
    message(FATAL_ERROR "Backend revision differs from docs/backend-revision.txt")
  endif()
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Backend static libraries" FORCE)
set(GGML_METAL ${FOUNDRY_METAL} CACHE BOOL "Metal" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU build" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Avoid undeclared OpenMP runtime" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/llama.cpp EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
add_library(foundry_json STATIC vendor/cjson/cJSON.c)
target_include_directories(foundry_json PUBLIC vendor/cjson)
add_library(foundry_core SHARED
  src/core/runtime.c src/core/session.c src/core/config.c src/core/metrics.c
  src/core/stream.c src/core/model.c src/core/registry.c src/core/gguf.c
  src/core/planner.c src/core/sha256.c src/backends/llama/adapter.cpp)
add_library(Foundry::foundry ALIAS foundry_core)
target_include_directories(foundry_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src/core src/backends)
target_compile_definitions(foundry_core PRIVATE _POSIX_C_SOURCE=200809L FOUNDRY_BACKEND_REVISION="${FOUNDRY_BACKEND_REVISION}")
target_link_libraries(foundry_core PRIVATE llama foundry_json Threads::Threads)
if(UNIX)
  target_link_libraries(foundry_core PRIVATE m)
endif()
set_target_properties(foundry_core PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1 EXPORT_NAME foundry)
add_executable(foundry src/cli/main.c)
target_compile_definitions(foundry PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(foundry PRIVATE foundry_core foundry_json Threads::Threads)
add_executable(foundry-example examples/basic.c)
target_link_libraries(foundry-example PRIVATE foundry_core)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
set_target_properties(foundry foundry-example PROPERTIES INSTALL_RPATH "${CMAKE_INSTALL_RPATH}")
foreach(t foundry_core foundry foundry-example)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  if(FOUNDRY_SANITIZE)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
if(FOUNDRY_BUILD_GUI)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
  add_executable(foundry-gui src/gui/main.c src/gui/runtime_worker.c src/gui/diagnostics.c src/gui/model_info.c)
  target_compile_definitions(foundry-gui PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(foundry-gui PRIVATE foundry_core foundry_json PkgConfig::GTK4)
  target_compile_options(foundry-gui PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(foundry-gui PROPERTIES INSTALL_RPATH "${CMAKE_INSTALL_RPATH}")
  install(TARGETS foundry-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS foundry_core EXPORT FoundryTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS foundry foundry-example RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/foundry DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES examples/basic.c DESTINATION ${CMAKE_INSTALL_DATADIR}/cinder-foundry/examples)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/cinder-foundry)
install(FILES vendor/llama.cpp/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/cinder-foundry/licenses RENAME llama.cpp-LICENSE)
install(FILES vendor/cjson/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/cinder-foundry/licenses RENAME cJSON-LICENSE)
install(EXPORT FoundryTargets NAMESPACE Foundry:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Foundry)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/FoundryConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/FoundryConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Foundry)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/FoundryConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/FoundryConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/FoundryConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Foundry)
set(CPACK_PACKAGE_NAME cinder-foundry)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR TGZ)
include(CPack)

if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_executable(foundry-test-core tests/unit/test_core.c)
  target_include_directories(foundry-test-core PRIVATE src/core)
  target_link_libraries(foundry-test-core PRIVATE foundry_core)
  add_executable(foundry-test-model tests/integration/test_model.c)
  target_link_libraries(foundry-test-model PRIVATE foundry_core)
  foreach(t foundry-test-core foundry-test-model)
    if(FOUNDRY_SANITIZE)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
  add_test(NAME core COMMAND foundry-test-core)
  add_test(NAME native-import-cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_cli.py $<TARGET_FILE:foundry>)
  if(UNIX)
    add_test(NAME clean-launcher COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/test_launcher.py ${CMAKE_CURRENT_SOURCE_DIR}/cinder.command)
    set_tests_properties(clean-launcher PROPERTIES TIMEOUT 15)
  endif()
  add_test(NAME version COMMAND foundry version)
  add_test(NAME sdk-empty-runtime COMMAND foundry-example)
  set_tests_properties(core native-import-cli version sdk-empty-runtime PROPERTIES TIMEOUT 60)
  set(FOUNDRY_TEST_MODEL "" CACHE FILEPATH "Optional local GGUF for real-model integration tests")
  if(FOUNDRY_TEST_MODEL)
    add_test(NAME real-model COMMAND foundry-test-model ${FOUNDRY_TEST_MODEL})
    set_tests_properties(real-model PROPERTIES TIMEOUT 180)
  endif()
  if(FOUNDRY_BUILD_GUI)
    add_test(NAME gui-version COMMAND foundry-gui --version)
    set_tests_properties(gui-version PROPERTIES TIMEOUT 15)
  endif()
endif()
if(BUILD_TESTING AND FOUNDRY_BUILD_GUI)
  add_executable(foundry-test-diagnostics tests/gui/test_diagnostics.c)
  target_link_libraries(foundry-test-diagnostics PRIVATE PkgConfig::GTK4)
  # This integration test requires a desktop session, like foundry-test-ui.
  add_executable(foundry-test-ui tests/gui/test_ui.c src/gui/runtime_worker.c src/gui/diagnostics.c src/gui/model_info.c)
  target_include_directories(foundry-test-ui PRIVATE src/gui)
  target_link_libraries(foundry-test-ui PRIVATE foundry_core foundry_json PkgConfig::GTK4)
  add_test(NAME gui-markdown COMMAND foundry-test-ui --markdown-only)
  set_tests_properties(gui-markdown PROPERTIES TIMEOUT 15)
  # Run foundry-test-ui directly in a desktop session for widget/keyboard checks.
  add_executable(foundry-test-worker tests/gui/test_worker.c src/gui/runtime_worker.c)
  target_include_directories(foundry-test-worker PRIVATE src/gui)
  target_link_libraries(foundry-test-worker PRIVATE foundry_core PkgConfig::GTK4)
  add_test(NAME gui-worker COMMAND foundry-test-worker)
  set_tests_properties(gui-worker PROPERTIES TIMEOUT 30)
  if(FOUNDRY_TEST_MODEL)
    add_test(NAME gui-worker-model COMMAND foundry-test-worker ${FOUNDRY_TEST_MODEL})
    set_tests_properties(gui-worker-model PROPERTIES TIMEOUT 180)
  endif()
endif()
if(FOUNDRY_BUILD_GUI)
  install(FILES packaging/io.github.cinderfoundry.desktop.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/cinder-foundry.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
  if(APPLE)
    add_executable(foundry-desktop MACOSX_BUNDLE src/gui/main.c src/gui/runtime_worker.c src/gui/diagnostics.c src/gui/model_info.c)
    target_link_libraries(foundry-desktop PRIVATE foundry_core foundry_json PkgConfig::GTK4)
    set_target_properties(foundry-desktop PROPERTIES OUTPUT_NAME foundry-gui
      MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in
      INSTALL_RPATH "@loader_path/../../../${CMAKE_INSTALL_LIBDIR}")
    install(TARGETS foundry-desktop BUNDLE DESTINATION .)
  endif()
endif()

foreach(gui_target foundry-gui foundry-desktop foundry-test-ui)
  if(TARGET ${gui_target})
    target_compile_definitions(${gui_target} PRIVATE FOUNDRY_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endforeach()
